## Acceptance Standard

The reconstructed app is not accepted until all gates pass:

- Resource parity: original manifest resource names equal rebuilt manifest resource names for the selected target binary.
- Dependency parity: all 36 original assembly references resolve to real .NET Framework/vendor/COM assemblies, not stubs.
- Build parity: Release x86 build succeeds on Windows using .NET Framework 4.7.2.
- PE/Win32 resource parity: rebuilt executables preserve original icon, version info, and Win32 resource footprint.
- Type/member coverage parity: every original TypeDef is present in source or explicitly classified as compiler/module/nested infrastructure with proof.
- Config parity: runtime `.config`, settings, connection strings, report paths, and registry/local configuration are recovered from the original deployment or explicitly documented as external blockers.
- Report parity: all `.RPT` basenames reachable by source literals are available in the deployed `Report` directory, or a written unreachable-proof exists per missing template.
- Startup parity: rebuilt app opens `MS_LOGIN`, successful login opens `MS_OPNCOMP`, company selection opens `MDI_FORM`.
- UI parity: representative DevExpress, DotNetBar, Qios, MSFlexGrid, and Crystal controls render with real controls, not fallback WinForms stubs.
- Workflow parity: master, transaction, report, print/export, backup/import, WhatsApp/SMS, RFID/tag, and Crystal report paths are exercised against the same database/configuration class as the original.

- [ ] **Step 1: Create the audit script**

```python
#!/usr/bin/env python3
import hashlib
import json
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
LOCAL_SITE_PACKAGES = Path("/Users/yakshitsavaliya/Library/Python/3.9/lib/python/site-packages")
if LOCAL_SITE_PACKAGES.exists():
    sys.path.insert(0, str(LOCAL_SITE_PACKAGES))

try:
    import dotnetfile
except Exception as exc:
    raise SystemExit(f"dotnetfile is required for metadata parity audit: {exc}")

TARGETS = {
    "DataCareNext": {
        "exe": ROOT / "DataCareNext.exe",
        "source_root": ROOT / "extracted_code" / "DataCareNext",
    },
    "NextDm": {
        "exe": ROOT / "NextDm.exe",
        "source_root": ROOT / "extracted_code" / "NextDm",
    },
}

def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def cs_files(source_root: Path):
    return sorted((source_root / "DataCareNext").glob("*.cs"))

def resx_resource_names(source_root: Path):
    names = {p.stem + ".resources" for p in source_root.glob("*.resx")}
    license_file = source_root / "DataCareNext.exe.licenses"
    if license_file.exists():
        names.add("DataCareNext.exe.licenses")
    return sorted(names)

def initialize_component_classes(source_root: Path):
    found = []
    for path in cs_files(source_root):
        if "InitializeComponent" in path.read_text(errors="ignore"):
            found.append(path.stem)
    return sorted(found)

def rpt_literals(source_root: Path):
    names = {}
    for path in cs_files(source_root):
        text = path.read_text(errors="ignore")
        for match in re.finditer(r'"([^"]+\.rpt)"', text, flags=re.I):
            raw = match.group(1)
            base = re.split(r"\\|/", raw)[-1]
            names.setdefault(base, str(path.relative_to(ROOT)))
    return dict(sorted(names.items(), key=lambda item: item[0].lower()))

def manifest_resources(exe: Path):
    pe = dotnetfile.DotNetPE(str(exe))
    resource_names = sorted(pe.ManifestResource.get_resource_names())
    subresource_counts = {}
    for item in pe.dotnet_resources:
        subresource_counts[item["Name"]] = item.get("NumberOfSubResources")
    return resource_names, subresource_counts

def assembly_refs(exe: Path):
    pe = dotnetfile.DotNetPE(str(exe))
    return sorted(pe.AssemblyRef.get_assemblyref_names())

def method_count(exe: Path):
    pe = dotnetfile.DotNetPE(str(exe))
    return len(pe.MethodDef.get_method_names())

def main():
    package_rpts = {p.name.lower(): p.name for p in ROOT.glob("*.rpt")}
    result = {
        "package": {
            "zip": str(ROOT / "Next_1106.zip"),
            "root_rpt_count": len(package_rpts),
            "root_rpts": sorted(package_rpts.values(), key=str.lower),
        },
        "targets": {},
    }
    for name, info in TARGETS.items():
        exe = info["exe"]
        source_root = info["source_root"]
        manifest, subcounts = manifest_resources(exe)
        source_resx = resx_resource_names(source_root)
        init_classes = initialize_component_classes(source_root)
        rpt_names = rpt_literals(source_root)
        present_rpts = sorted([r for r in rpt_names if r.lower() in package_rpts], key=str.lower)
        missing_rpts = sorted([r for r in rpt_names if r.lower() not in package_rpts], key=str.lower)
        result["targets"][name] = {
            "exe_size": exe.stat().st_size,
            "exe_sha256": sha256(exe),
            "assembly_refs": assembly_refs(exe),
            "assembly_ref_count": len(assembly_refs(exe)),
            "metadata_method_count": method_count(exe),
            "manifest_resource_count": len(manifest),
            "manifest_resources": manifest,
            "source_resx_count_including_license": len(source_resx),
            "missing_source_resources": sorted(set(manifest) - set(source_resx)),
            "extra_source_resources": sorted(set(source_resx) - set(manifest)),
            "subresource_counts_for_missing_source_resources": {
                key: subcounts.get(key) for key in sorted(set(manifest) - set(source_resx))
            },
            "initialize_component_class_count": len(init_classes),
            "rpt_literal_count": len(rpt_names),
            "rpt_present_in_package_count": len(present_rpts),
            "rpt_missing_from_package_count": len(missing_rpts),
            "rpt_missing_from_package": {name: rpt_names[name] for name in missing_rpts},
        }
    out_dir = ROOT / "audit"
    out_dir.mkdir(exist_ok=True)
    out_path = out_dir / "next1106-parity.json"
    out_path.write_text(json.dumps(result, indent=2), encoding="utf-8")
    print(out_path)

if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
import json
import re
import sys
from collections import Counter, defaultdict
from pathlib import Path

if len(sys.argv) != 2:
    raise SystemExit("usage: parse_build_errors.py BUILD_LOG")

log_path = Path(sys.argv[1])
lines = log_path.read_text(errors="ignore").splitlines()
errors = []
warnings = []

for line in lines:
    error_match = re.search(r"(?P<file>.*?\.(?:cs|resx|csproj))\((?P<line>\d+),(?P<col>\d+)\): error (?P<code>CS\d+|MSB\d+|NETSDK\d+): (?P<message>.*?)(?: \[.*\])?$", line)
    if error_match:
        errors.append(error_match.groupdict())
        continue
    warning_match = re.search(r"(?P<file>.*?\.(?:cs|resx|csproj))\((?P<line>\d+),(?P<col>\d+)\): warning (?P<code>CS\d+|MSB\d+|NETSDK\d+): (?P<message>.*?)(?: \[.*\])?$", line)
    if warning_match:
        warnings.append(warning_match.groupdict())

missing_types = Counter()
files_by_code = defaultdict(Counter)
for item in errors:
    code = item["code"]
    files_by_code[code][item["file"]] += 1
    missing = re.search(r"The type or namespace name '([^']+)' could not be found", item["message"])
    if missing:
        missing_types[missing.group(1)] += 1

summary_match = None
for line in lines:
    m = re.search(r"^\s*(\d+)\s+Error\(s\)", line)
    if m:
        summary_match = int(m.group(1))

result = {
    "log": str(log_path),
    "parsed_error_lines": len(errors),
    "msbuild_summary_errors": summary_match,
    "total_errors": summary_match if summary_match is not None else len(errors),
    "parsed_warning_lines": len(warnings),
    "errors_by_code": Counter(item["code"] for item in errors),
    "missing_types": missing_types,
    "top_files_by_code": {
        code: dict(counter.most_common(20)) for code, counter in files_by_code.items()
    },
}

print(json.dumps(result, indent=2))
raise SystemExit(1 if result["total_errors"] else 0)
```

- [ ] **Step 5: Create the `CS0621` mechanical fix script**

This script converts only the specific decompiler artifact shape from field-targeted auto-properties to explicit backing fields. It preserves property names, access modifiers, `virtual`, setter synchronization, and `AccessedThroughProperty` metadata.

```python
#!/usr/bin/env python3
import re
import sys
from pathlib import Path

if len(sys.argv) < 2:
    raise SystemExit("usage: fix_field_accessed_auto_properties.py FILE_OR_DIR [FILE_OR_DIR...]")

PROPERTY_RE = re.compile(
    r'(?P<indent>[ \t]*)\[field: AccessedThroughProperty\("(?P<accessed>[^"]+)"\)\]\n'
    r'(?P=indent)(?P<mods>(?:public|private|internal|protected|static|virtual|override|sealed|new|readonly|unsafe|extern|partial|async|[ \t])+?)'
    r'(?P<type>[A-Za-z_][A-Za-z0-9_<>,\.\?\[\] ]*[A-Za-z0-9_\]\?])\s+'
    r'(?P<name>[A-Za-z_][A-Za-z0-9_]*)\n'
    r'(?P=indent)\{\n'
    r'(?P=indent)\tget; \[MethodImpl\(MethodImplOptions\.Synchronized\)\]\n'
    r'(?P=indent)\tset;\n'
    r'(?P=indent)\}',
    re.MULTILINE,
)

def backing_name(name: str) -> str:
    return "_" + name if not name.startswith("_") else "_" + name

def convert(match: re.Match) -> str:
    indent = match.group("indent")
    accessed = match.group("accessed")
    mods = " ".join(match.group("mods").split())
    type_name = " ".join(match.group("type").split())
    prop_name = match.group("name")
    field_name = backing_name(prop_name)
    return (
        f'{indent}[CompilerGenerated]\n'
        f'{indent}[DebuggerBrowsable(DebuggerBrowsableState.Never)]\n'
        f'{indent}[AccessedThroughProperty("{accessed}")]\n'
        f'{indent}private {type_name} {field_name};\n\n'
        f'{indent}{mods} {type_name} {prop_name}\n'
        f'{indent}{{\n'
        f'{indent}\tget\n'
        f'{indent}\t{{\n'

    if path.is_dir():

    else:

    text = path.read_text(errors="ignore")

    if count:

        changed += count
```

```python
#!/usr/bin/env python3
import json
import sys
from pathlib import Path

LOCAL_SITE_PACKAGES = Path("/Users/yakshitsavaliya/Library/Python/3.9/lib/python/site-packages")
if LOCAL_SITE_PACKAGES.exists():
    sys.path.insert(0, str(LOCAL_SITE_PACKAGES))

import dotnetfile

def fixed_version(pe):
    info = pe.VS_FIXEDFILEINFO[0]
    file_ms = info.FileVersionMS
    file_ls = info.FileVersionLS
    product_ms = info.ProductVersionMS
    product_ls = info.ProductVersionLS
    return {
        "file": [
            (file_ms >> 16) & 0xFFFF,
            file_ms & 0xFFFF,
            (file_ls >> 16) & 0xFFFF,
            file_ls & 0xFFFF,
        ],
        "product": [
            (product_ms >> 16) & 0xFFFF,
            product_ms & 0xFFFF,
            (product_ls >> 16) & 0xFFFF,
            product_ls & 0xFFFF,
        ],
    }

def identity(path):
    pe = dotnetfile.DotNetPE(str(path))
    assembly_version = pe.Assembly.get_assembly_version_information()
    return {
        "path": str(path),
        "assembly_name": pe.Assembly.get_assembly_name(),
        "assembly_version": [
            assembly_version.MajorVersion,
            assembly_version.MinorVersion,
            assembly_version.BuildNumber,
            assembly_version.RevisionNumber,
        ],
        "assembly_refs": pe.AssemblyRef.get_assemblyref_names(),
        "win32_resource_entry_count": getattr(pe, "_PE__total_resource_entries_count", None),
        "win32_resource_bytes": getattr(pe, "_PE__total_resource_bytes", None),
        "fixed_version": fixed_version(pe),
        "streams": pe.get_stream_names(),
    }

if len(sys.argv) != 3:
    raise SystemExit("usage: compare_pe_identity.py ORIGINAL_EXE REBUILT_EXE")

original = identity(Path(sys.argv[1]))
rebuilt = identity(Path(sys.argv[2]))
checks = {
    "assembly_name": original["assembly_name"] == rebuilt["assembly_name"],
    "assembly_version": original["assembly_version"] == rebuilt["assembly_version"],
    "assembly_refs": original["assembly_refs"] == rebuilt["assembly_refs"],
    "win32_resource_entry_count": original["win32_resource_entry_count"] == rebuilt["win32_resource_entry_count"],
    "win32_resource_bytes": original["win32_resource_bytes"] == rebuilt["win32_resource_bytes"],
    "fixed_version": original["fixed_version"] == rebuilt["fixed_version"],
}
result = {"original": original, "rebuilt": rebuilt, "checks": checks}
print(json.dumps(result, indent=2))
raise SystemExit(0 if all(checks.values()) else 1)
```

```python
#!/usr/bin/env python3
import json
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
LOCAL_SITE_PACKAGES = Path("/Users/yakshitsavaliya/Library/Python/3.9/lib/python/site-packages")
if LOCAL_SITE_PACKAGES.exists():
    sys.path.insert(0, str(LOCAL_SITE_PACKAGES))

import dotnetfile

TARGETS = {
    "DataCareNext": {
        "exe": ROOT / "DataCareNext.exe",
        "source_roots": [
            ROOT / "extracted_code" / "DataCareNext" / "DataCareNext",
            ROOT / "extracted_code" / "DataCareNext" / "DataCareNext.My",
            ROOT / "extracted_code" / "DataCareNext" / "DataCareNext.My.Resources",
            ROOT / "extracted_code" / "DataCareNext",
            ROOT / "extracted_code" / "DataCareNext" / "Properties",
        ],
    },
    "NextDm": {
        "exe": ROOT / "NextDm.exe",
        "source_roots": [
            ROOT / "extracted_code" / "NextDm" / "DataCareNext",
            ROOT / "extracted_code" / "NextDm" / "DataCareNext.My",
            ROOT / "extracted_code" / "NextDm" / "DataCareNext.My.Resources",
            ROOT / "extracted_code" / "NextDm",
            ROOT / "extracted_code" / "NextDm" / "Properties",
        ],
    },
}

DECL_RE = re.compile(r"\b(?:class|struct|interface|enum|delegate)\s+([A-Za-z_][A-Za-z0-9_]*)")

def metadata_types(exe):
    pe = dotnetfile.DotNetPE(str(exe))
    names = set(pe.TypeDef.get_type_names())
    return sorted(name for name in names if name != "<Module>")

def source_types(source_roots):
    names = set()
    files_by_name = {}
    for root in source_roots:
        if root.is_file() and root.suffix == ".cs":
            paths = [root]
        elif root.is_dir():
            paths = sorted(root.glob("*.cs"))
        else:
            paths = []
        for path in paths:
            text = path.read_text(errors="ignore")
            for match in DECL_RE.finditer(text):
                name = match.group(1)
                names.add(name)
                files_by_name.setdefault(name, str(path.relative_to(ROOT)))
    return names, files_by_name

def main():
    result = {}
    failed = False
    for target, info in TARGETS.items():
        metadata = set(metadata_types(info["exe"]))
        source, files = source_types(info["source_roots"])
        missing = sorted(metadata - source)
        extra = sorted(source - metadata)
        result[target] = {
            "metadata_type_count": len(metadata),
            "source_type_count": len(source),
            "missing_metadata_types_in_source": missing,
            "extra_source_types_not_in_metadata": extra,
            "source_files_by_type": {name: files[name] for name in sorted(files)},
        }
        failed = failed or bool(missing)
    out_dir = ROOT / "audit"
    out_dir.mkdir(exist_ok=True)
    out_path = out_dir / "type-coverage.json"
    out_path.write_text(json.dumps(result, indent=2), encoding="utf-8")
    print(out_path)
    raise SystemExit(1 if failed else 0)

if __name__ == "__main__":
    main()
```
